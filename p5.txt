Write a Lex program that distinguishes keywords, integers, floats,
identifiers, operators, and comments in any simple programming language.

==============================================================================
CODE  (p5.l)
==============================================================================

%{
#include <stdio.h>
%}
%%
"int"|"float"|"char"|"double"|"void"|"if"|"else"|"while"|"for"|"do"|"return"|"switch"|"case"|"break"|"continue"   { printf("%-30s KEYWORD\n", yytext); }
"//".*                                  { printf("%-30s COMMENT (single line)\n", yytext); }
"/*"([^*]|\*+[^*/])*\*+"/"              { printf("%-30s COMMENT (multi line)\n", yytext); }
[0-9]+\.[0-9]+                          { printf("%-30s FLOAT\n", yytext); }
[0-9]+                                  { printf("%-30s INTEGER\n", yytext); }
[a-zA-Z_][a-zA-Z0-9_]*                  { printf("%-30s IDENTIFIER\n", yytext); }
"=="|"!="|"<="|">="|"&&"|"||"|"++"|"--"|"+"|"-"|"*"|"/"|"%"|"="|"<"|">"   { printf("%-30s OPERATOR\n", yytext); }
[ \t\n]+                                { }
.                                       { printf("%-30s SEPARATOR / OTHER\n", yytext); }
%%
int yywrap(void) { return 1; }
int main(int argc, char *argv[])
{
    if (argc > 1) {
        yyin = fopen(argv[1], "r");
        if (yyin == NULL) {
            printf("Cannot open file %s\n", argv[1]);
            return 1;
        }
    }
    yylex();
    return 0;
}

==============================================================================
COMMANDS TO COMPILE AND RUN
==============================================================================

flex p5.l
gcc lex.yy.c -o p5
./p5 sample.c

(If your lab uses the classic lex tool, use `lex` in place of `flex`.)

==============================================================================
SAMPLE INPUT AND EXPECTED OUTPUT
==============================================================================

Input file (sample.c):
----------------------------------------
// sample program
int main() {
    float x = 3.14;
    int y = 10; /* counter */
    if (x >= y) y++;
    return 0;
}
----------------------------------------
