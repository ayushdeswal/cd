A program in Lex to recognize a valid arithmetic expression.

==============================================================================
CODE  (p8.l)
==============================================================================

%{
#include <stdio.h>
int expect_operand = 1;  /* 1: operand or '(' expected; 0: operator or ')' expected */
int paren = 0;
int valid = 1;
%}
%%
[a-zA-Z_][a-zA-Z0-9_]*|[0-9]+(\.[0-9]+)?   { if (!expect_operand) valid = 0; expect_operand = 0; }
[-+*/]      { if (expect_operand) valid = 0; expect_operand = 1; }
"("         { if (!expect_operand) valid = 0; paren++; expect_operand = 1; }
")"         { if (expect_operand) valid = 0; paren--; if (paren < 0) valid = 0; expect_operand = 0; }
[ \t]       { }
\n          { return 0; }
.           { valid = 0; }
%%
int yywrap(void) { return 1; }
int main()
{
    printf("Enter an expression: ");
    yylex();
    if (valid && paren == 0 && !expect_operand)
        printf("Valid expression\n");
    else
        printf("Invalid expression\n");
    return 0;
}

==============================================================================
COMMANDS TO COMPILE AND RUN
==============================================================================

flex p8.l
gcc lex.yy.c -o p8
./p8

(If your lab uses the classic lex tool, use `lex` in place of `flex`.)

SAMPLE INPUT AND EXPECTED OUTPUT
==============================================================================

$ ./p8

$ ./p8

Valid expression
