Write a Lex program to count the number of words, characters, blank spaces and
lines in a C file.

==============================================================================
CODE  (p6.l)
==============================================================================

%{
#include <stdio.h>
int words = 0, chars = 0, spaces = 0, lines = 0;
%}
%%
\n          { lines++; chars++; }
[ \t]       { spaces++; chars++; }
[^ \t\n]+   { words++; chars += yyleng; }
%%
int yywrap(void) { return 1; }
int main(int argc, char *argv[])
{
    if (argc > 1) {
        yyin = fopen(argv[1], "r");
        if (yyin == NULL) {
            printf("Cannot open file %s\n", argv[1]);
            return 1;
        }
    }
    yylex();
    printf("Number of words        = %d\n", words);
    printf("Number of characters   = %d\n", chars);
    printf("Number of blank spaces = %d\n", spaces);
    printf("Number of lines        = %d\n", lines);
    return 0;
}

==============================================================================
COMMANDS TO COMPILE AND RUN
==============================================================================

flex p6.l
gcc lex.yy.c -o p6
./p6 sample.c

(If your lab uses the classic lex tool, use `lex` in place of `flex`.)

SAMPLE INPUT AND EXPECTED OUTPUT

int main() {
    float x = 3.14;
    int y = 10; /* counter */
    if (x >= y) y++;
    return 0;
}
