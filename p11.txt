Program in YACC to recognize the string 'abbb', 'ab', 'a' of the language (a^n
b^n, n>=1).

==============================================================================
CODE  (p11.y)
==============================================================================

%{
#include <stdio.h>
int yylex(void);
void yyerror(const char *s);
%}
%token A B
%%
Start : S '\n'      { printf("Valid string\n"); }
      ;
S     : A S B
      | A B
      ;
%%
int yylex(void)
{
    int c = getchar();
    if (c == 'a') return A;
    if (c == 'b') return B;
    if (c == EOF) return 0;
    return c;
}
void yyerror(const char *s) { printf("Invalid string\n"); }
int main()
{
    printf("Enter a string: ");
    yyparse();
    return 0;
}

==============================================================================
COMMANDS TO COMPILE AND RUN
==============================================================================

bison -y p11.y
gcc y.tab.c -o p11
./p11

(If your lab uses the classic yacc tool, use `yacc` in place of `bison -y`.)
